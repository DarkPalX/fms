QUERY:
    SELECT 
        fms.*,
        wfs.id AS transaction_id,
        wfs.id,
        approval.status AS approval_status,
        users.username AS approver_username
    FROM 
        fms_db.dbo.issuance_request_tbl fms
    JOIN 
        [PMC-WFS].dbo.transactions wfs 
            ON CAST(wfs.id AS VARCHAR) = fms.wfs_transaction_id
    JOIN 
        [PMC-WFS].dbo.approval_status approval 
            ON approval.transaction_id = wfs.id
    JOIN 
        [PMC-WFS].dbo.users users 
            ON users.id = approval.approver_id
    WHERE 
        wfs_status = 'FULLY APPROVED' AND fms.status = 'PENDING'
